\subsection{Detailed Specifications}

\subsubsection{UC-01: Log Workout Set and Compute Progression}
\begin{table}[h!]
\centering
\small
\begin{tabularx}{\textwidth}{l X}
\toprule
\textbf{Use Case} & \textbf{UC-01: Log Workout Set and Compute Automatic Progression} \\
\midrule
\textbf{Primary Actor} & Athlete / End User \\
\textbf{Preconditions} & At least one exercise exists in the catalog and an active workout session is started. \\
\textbf{Main Flow} & 
1. The user selects the current exercise in the active session.\newline
2. The system displays target weight ($kg$) and reps calculated by UC-02.\newline
3. The user inputs load, completed reps, and RPE ($6.0 - 10.0$).\newline
4. The user confirms set registration.\newline
5. The system calculates estimated 1RM, persists the set to SQLite, and updates session effective volume. \\
\textbf{Alternative Flow} & 
\textit{3a. User marks set as Warmup Set:}\newline
- System records the set but excludes it from effective volume and 1RM calculation. \\
\textbf{Postconditions} & Set recorded in SQLite with calculated 1RM. \\
\bottomrule
\end{tabularx}
\end{table}

\subsubsection{UC-02: Get Automatic Overload Recommendation}
\begin{table}[h!]
\centering
\small
\begin{tabularx}{\textwidth}{l X}
\toprule
\textbf{Use Case} & \textbf{UC-02: Get Automatic Overload Recommendation} \\
\midrule
\textbf{Primary Actor} & Athlete / End User \\
\textbf{Preconditions} & Exercise selected from catalog. \\
\textbf{Main Flow} & 
1. System fetches last working sets logged for the exercise.\newline
2. System calculates average $RIR$ and completion rate against target rep range.\newline
3. System applies load rule ($+2.5\,kg$ compound / $+1.25\,kg$ isolation / $MAINTAIN$ / $DELOAD$).\newline
4. System presents recommended load and target reps with mathematical justification. \\
\textbf{Postconditions} & Recommendation generated deterministically without side effects. \\
\bottomrule
\end{tabularx}
\end{table}

\subsubsection{UC-04: Finish \& Finalize Workout Session}
\begin{table}[h!]
\centering
\small

\toprule
\textbf{Use Case} & \textbf{UC-04: Finish \& Finalize Workout Session} \\
\midrule
\textbf{Primary Actor} & Athlete / End User \\
\textbf{Preconditions} & Active workout session in progress. \\
\textbf{Main Flow} & 
1. User taps "Finish Workout".\newline
2. System calculates session duration, total tonnage, and total effective volume.\newline
3. System identifies any Personal Records (PRs) set during the session.\newline
