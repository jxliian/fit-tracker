\section{UML Use Cases}

\subsection{System Actors}
\begin{itemize}
    \item \textbf{Athlete / End User:} Individual executing workout routines, logging sets, and monitoring overload metrics.
\end{itemize}

\subsection{Detailed Use Case Specification UC-01}

\begin{table}[h!]
\centering
\caption{Use Case UC-01: Log Workout Set and Compute Progression}
\begin{tabularx}{\textwidth}{l X}
\toprule
\textbf{Use Case} & \textbf{UC-01: Log Workout Set and Compute Automatic Progression} \\
\midrule
\textbf{Primary Actor} & Athlete / End User \\
\textbf{Preconditions} & At least one exercise exists in the catalog and an active workout session is started. \\
\textbf{Main Flow} & 
1. The user selects the current exercise in the active session.\newline
2. The system displays previously calculated target weight and reps.\newline
3. The user inputs performed load ($kg$), completed reps, and RPE ($6.0 - 10.0$).\newline
4. The user taps "Complete Set".\newline
5. The system persists the set to SQLite, starts the rest timer, and increments set counter.\newline
6. The overload engine updates the session's estimated 1RM. \\
\textbf{Alternative Flow} & 
\textit{3a. User marks set as "Warmup Set":}\newline
- System saves the set but excludes it from effective volume and record 1RM calculations.\newline
\textit{3b. User indicates muscle failure (RPE 10 / RIR 0):}\newline
- System adjusts fatigue accumulation factor for the next session calculation. \\
\textbf{Postconditions} & Set persisted in SQLite; rest timer triggers audio/vibration notification upon expiration. \\
\bottomrule
\end{tabularx}
\end{table}
